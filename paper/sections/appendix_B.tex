\section{Proofs}
\label{app:proofs}

Notation and the sign convention are those of the main text; $m\equiv\alpha_T(1-\alpha_D)>0$ at the symmetric benchmark. Unless stated otherwise, derivatives are first-order and evaluated at a balanced free-trade baseline, using the linear system of \cref{app:linear}.

\subsection{Multilateral adjustment and symmetric partners}

\begin{proof}[Proof of \cref{lem:gs}]
Free trade is not required. By homogeneity and Euler's theorem, $\sum_{l\in\mathcal I}\widetilde J_{il}=\TB_i=0$ at the balanced baseline. Since $\nu_l=\chi_l-\chi_A$ with $\chi_A$ fixed, $J$ is $\widetilde J$ with row and column $A$ deleted. Its off-diagonal entries are nonnegative, and for each $i\in\mathcal I_{-A}$
\begin{equation*}
|J_{ii}|-\sum_{l\in\mathcal I_{-A},\,l\neq i}J_{il}=\widetilde J_{iA}\geq0,
\end{equation*}
strictly in at least one row. The diagonal of $-J$ is strictly positive: for $N=2$, $|J_{BB}|=\widetilde J_{BA}>0$; for $N\geq3$, $J_{ii}=0$ would force the nonnegative off-diagonal entries of row $i$ to vanish, contradicting irreducibility. Write $-J=D-K$ with $D=\operatorname{diag}(-J)>0$ and $K\geq0$. The matrix $D^{-1}K$ is nonnegative and irreducible, with every row sum at most one and at least one strictly below one. The spectral radius of an irreducible nonnegative matrix is at most its largest row sum, with equality only if all row sums are equal (Perron--Frobenius), so the spectral radius of $D^{-1}K$ is below one and
\begin{equation*}
(-J)^{-1}=(I-D^{-1}K)^{-1}D^{-1}=\sum_{k\geq0}(D^{-1}K)^kD^{-1}\geq0 .
\end{equation*}
Hence MML holds. Because $D^{-1}K$ is irreducible, the series is entrywise positive, and $\det(-J)=\det D\cdot\det(I-D^{-1}K)>0$.
\end{proof}

\begin{proof}[Proof of \cref{prop:decomposition}]
No demand system, free-trade baseline or stability condition is used. If $PMP'=M$ for every $(N-1)\times(N-1)$ permutation matrix $P$, conjugating by permutations shows that all diagonal entries are equal and all off-diagonal entries are equal, so $M=aI+b\bone\bone'$, with eigenvalue $a+(N-1)b$ on $\operatorname{span}\{\bone\}$ and $a$ on the zero-sum subspace. For $-J=a_JI+b_J\bone\bone'$ and $G^{(A)}=a_GI+b_G\bone\bone'$ this gives $j_S=a_J+(N-1)b_J$, $j_D=a_J$, $g_S=a_G+(N-1)b_G$ and $g_D=a_G$, and for $j_S,j_D\neq0$
\begin{equation*}
(-J)^{-1}=\frac{1}{j_D}I+\frac{1}{N-1}\Big(\frac{1}{j_S}-\frac{1}{j_D}\Big)\bone\bone'.
\end{equation*}
Thus $(-J)^{-1}$ and $G^{(A)}$ share the two eigenspaces, and applying them to $t=\bar t\bone+\widetilde t$ gives \cref{eq:decomp}. Partner symmetry makes the baseline flows $f_{Aj}+f_{jA}$ equal across $j$, so $w_{Aj}=1/(N-1)$; since $\widetilde t$ sums to zero, $d\nu^E_A=\lambda_{\mathrm{avg}}\bar t$, and the other two parts of \cref{eq:decompparts} follow directly.

For the MML characterization, the off-diagonal entry of $J$ is $-b_J=(j_D-j_S)/(N-1)$, so the sign pattern holds if and only if $j_D\geq j_S$. If $j_D\geq j_S>0$, $-J$ is nonsingular and every entry of $(-J)^{-1}$ above is nonnegative, so MML holds. Conversely, under MML the sign pattern gives $j_D\geq j_S$, nonsingularity gives $j_S\neq0$, and $(-J)^{-1}\bone=\bone/j_S\geq0$ gives $j_S>0$.
\end{proof}

\subsection{The symmetric nested-CES model}

The baseline is the symmetric benchmark of \cref{sec:model} with $N\geq3$; for $N=2$, \cref{thm:main} reduces to \cref{eq:twocountry}.

\begin{proof}[Proof of \cref{thm:main}]
The benchmark is invariant to relabeling $A$'s partners, so \cref{prop:decomposition} applies, and the eigenvalues \cref{eq:eigenvalues} are those of \cref{lem:jacobian} and \cref{eq:geigen}. Since $m>0$, $j_S=mM_N/(N-1)>0$ if and only if $M_N>0$, and then $j_D=Nj_S>j_S$. By \cref{prop:decomposition}, MML is therefore equivalent to $M_N>0$; if $M_N\leq0$, then $j_S\leq0$ and MML fails. If $\eta\geq1$, the bracket in \cref{eq:MN} is $1$ plus a nonnegative term plus $(N-2)\rho>0$, so $M_N>0$. Given $M_N>0$, dividing \cref{eq:geigen} by the eigenvalues of $-J$ gives $\lambda_{\mathrm{avg}}=-\rho_D/M_N$ and $\lambda_{\mathrm{rel}}=-\rho/(NM_N)$, and \cref{eq:lambdas} is \cref{eq:decomp}.
\end{proof}

\begin{proof}[Proof of \cref{prop:unilateral}]
\Cref{eq:unilateral} follows from \cref{eq:lambdas} (\cref{app:linear-unilateral}). Under MML $M_N>0$, and $\rho_D>0$ since $\alpha_D\eta>0$; hence $d\nu^E_A/d\tau<0$ and, since $N\rho_D+(N-2)\rho>0$, $d\nu_B/d\tau<0$. The denominator of $d\nu_c/d\tau$ is positive, so $d\nu_c/d\tau>0$ if and only if $\rho>N\rho_D$, with $d\nu_c=0$ at $\rho=N\rho_D$. For (iv), with $k\leq N-2$ targets the response against each untargeted partner is $k$ times the unilateral response (\cref{app:linear-unilateral}), so the threshold is again $\rho>N\rho_D$. For a general tariff vector with $\bar t>0$, \cref{eq:reltreat} gives the stated condition; when $t_j\geq0$ its left side is nonnegative, so it can hold only if $1-N\rho_D/\rho>0$, that is, $\rho>N\rho_D$, and it then implies $t_j<\bar t$.
\end{proof}

\begin{proof}[Proof of \cref{cor:single}]
Since $N\rho_D=N[1+\alpha_D(\eta-1)-\alpha_T(1-\alpha_D)]$, \cref{eq:single} is an identity; under $\alpha_D\geq1/N$ both terms on its right are nonnegative, so $N\rho_D\geq\eta$, and since a bystander depreciation requires $\rho>N\rho_D$ (\cref{prop:unilateral}), it requires $\rho>\eta$. With $\rho=\eta=\sigma$, \cref{eq:unilateral} gives $d\nu_c/d\tau=(\sigma-N\rho_D)/[N(N-1)M_N]$ with $\sigma-N\rho_D=-[\sigma(N\alpha_D-1)+N(1-\alpha_D)(1-\alpha_T)]\leq0$; MML is used only to give $M_N>0$, so $d\nu_c/d\tau\leq0$. Since $\sigma>0$ and $\alpha_D<1$, the bracket vanishes if and only if $\alpha_D=1/N$ and $\alpha_T=1$. At $\alpha_D=1/N$, $N\rho_D-\sigma=(N-1)(1-\alpha_T)$ and $M_N=\sigma>0$, so MML holds for every $\sigma>0$, and $d\nu_c/d\tau=-(1-\alpha_T)/(N\sigma)$, which is negative and tends to zero as $\sigma\to\infty$ if $\alpha_T<1$ and is identically zero if $\alpha_T=1$.
\end{proof}

\Cref{prop:composite} applies to any marginal tariff change that is invariant to permutations of the $N-2$ bystanders and leaves tariffs among bystanders unchanged; the composite's tariffs are the common bystander values. In the three-region economy, $C$ has income $y_C=N-2$, home share $h_C$ with $(N-1)h_C=N-3+2\alpha_D$, weight $\tfrac12$ on each of $A$ and $B$ in its import nest, tradable share $\alpha_T$, cross-origin elasticity $\rho$ and home--import elasticity $\eta_C$; $A$ and $B$ place import weight $\frac{1}{N-1}$ on each other and $\frac{N-2}{N-1}$ on $C$.

\begin{proof}[Proof of \cref{prop:composite}]
The argument compares the two linearized systems at the benchmark; it does not claim that the nonlinear economies coincide at finite tariffs.

\emph{Common bystander response.} By \cref{lem:jacobian}, $-J=j_S(NI-\bone\bone')$ with $j_S=mM_N/(N-1)\neq0$, so \cref{eq:JG} has a unique solution. Baseline and shock are invariant to permutations of the bystanders, hence so is the solution: $d\nu_c=d\nu_C$ for all $c$. The equilibrium is then given by $d\TB_B=0$ and $\sum_cd\TB_c=0$ in $(d\nu_B,d\nu_C)$.

\emph{Composite structure.} Summing \cref{eq:tb} over bystanders, intra-bloc flows cancel, so $\sum_c\TB_c$ has the form of $\TB_C=f_{AC}+f_{BC}-f_{CA}-f_{CB}$. The mapped income, home share and import weights reproduce the baseline flows between the bloc and each of $A$ and $B$, and since all bystander goods face a common price change in $A$ and in $B$, $A$'s and $B$'s demand for the bloc's goods responds as their demand for $C$'s good. It remains to match the bloc's demand for $A$'s and $B$'s goods.

\emph{Internal elasticity.} Measure prices in bystander currency, so other bystanders' goods have $dp=0$, and write $S\equiv dp_{cA}+dp_{cB}$ and $D\equiv dp_{cA}-dp_{cB}$. A bystander's import index moves by $S/(N-1)$ and the composite's by $S/2$, so
\begin{equation*}
\begin{aligned}
d\log s_{cA}&=(1-\rho)\frac D2+\Big[(1-\rho)\frac{N-3}{2(N-1)}+(1-\eta)\frac{\alpha_D}{N-1}\Big]S,\\
d\log s_{CA}&=(1-\rho)\frac D2+(1-\eta_C)h_C\frac S2.
\end{aligned}
\end{equation*}
The $D$ terms coincide; the $S$ terms coincide if and only if $(1-\eta_C)(N-1)h_C=(1-\rho)(N-3)+2\alpha_D(1-\eta)$, which, with $(N-1)h_C=N-3+2\alpha_D$, is \cref{eq:etaC}. The argument for $B$'s good replaces $D$ by $-D$.

\emph{Equivalence.} The two equilibrium conditions, including tariff terms, are therefore identical to the composite's. Their restricted Jacobian has determinant $j_S^2N(N-2)\neq0$, so both systems have the same unique solution for $(d\nu_B,d\nu_C)$, and the composite's trade weights $\frac{1}{N-1}$ and $\frac{N-2}{N-1}$ reproduce $A$'s NEER. For $N=3$, $h_C=\alpha_D$ and $\eta_C=\eta$, so $C$ is the single bystander. The flow-by-flow verification and the restricted system are in the online supplement.
\end{proof}

\subsection{Retaliation, welfare, and asymmetry}

\begin{proof}[Proof of \cref{prop:retaliation}]
Under MML $M_N>0$ (\cref{thm:main}), and the response is linear in tariffs, so responses to simultaneous tariffs superpose. (i) Shock~1, $d\tau_{AB}=d\tau$, is \cref{eq:unilateral}. Shock~2, $d\tau_{BA}=d\tau$, is the same experiment with $A$ and $B$ exchanged, so $d\log e_{BA}=-\frac{N\rho_D+(N-2)\rho}{N(N-1)M_N}d\tau$ and $d\log e_{Bc}=\frac{\rho-N\rho_D}{N(N-1)M_N}d\tau$; with $e_{AB}=1/e_{BA}$ and $e_{Ac}=e_{AB}e_{Bc}$,
\begin{equation*}
d\nu^{(2)}_B=\frac{N\rho_D+(N-2)\rho}{N(N-1)M_N}\,d\tau,\qquad d\nu^{(2)}_c=\frac{\rho}{NM_N}\,d\tau.
\end{equation*}
Adding shock~1, $d\nu_B=0$ and
\begin{equation*}
\frac{d\nu_c}{d\tau}=\frac{\rho-N\rho_D}{N(N-1)M_N}+\frac{\rho}{NM_N}=\frac{\rho-\rho_D}{(N-1)M_N}
\end{equation*}
for every bystander, so cross rates among bystanders are unchanged and $d\log e_{Bc}=d\nu_c-d\nu_B=d\nu_c$. With equal weights, $d\nu^E_A=\frac{N-2}{N-1}d\nu_c$, and $B$'s NEER, $[0+(N-2)d\nu_c]/(N-1)$, is the same. Since $M_N>0$ and $N\geq3$, every response has the sign of $\rho-\rho_D$.

(ii) $A$'s uniform tariff contributes $\lambda_{\mathrm{avg}}d\tau=-\frac{\rho_D}{M_N}d\tau$ to every $d\nu_j$. Partner $j$'s tariff on $A$ is shock~2 with $j$ in the role of $B$: it contributes $\frac{N\rho_D+(N-2)\rho}{N(N-1)M_N}d\tau$ to $d\nu_j$ and $\frac{\rho}{NM_N}d\tau$ to each of the other $N-2$ partners. Summing,
\begin{equation*}
\frac{d\nu_j}{d\tau}=-\frac{\rho_D}{M_N}+\frac{N\rho_D+(N-2)\rho}{N(N-1)M_N}+\frac{(N-2)\rho}{NM_N}=\frac{(N-2)(\rho-\rho_D)}{(N-1)M_N},
\end{equation*}
common to all $j$, so cross rates among partners are unchanged and $d\nu^E_A$ equals this response, with the sign of $\rho-\rho_D$.
\end{proof}

\begin{proof}[Proof of \cref{lem:welfare}]
Indirect utility is $V_A=\kappa\,I_A/(P_{TA}^{\alpha_T}P_{NA}^{1-\alpha_T})$, so the welfare change in units of income is $d\log I_A-\alpha_T\,d\log P_{TA}-(1-\alpha_T)\,d\log P_{NA}$. By Shephard's lemma, $\alpha_T\,d\log P_{TA}=\sum_js_{Aj}\,d\log p_{Aj}$. Under the producer-price normalization $P_{NA}$ and the domestic tradable's price are fixed in $A$'s currency, and at free trade $d\log p_{Aj}=d\nu_j+d\tau_{Aj}$. Income is $I_A=w_AL_A/(1-\sum_j\theta_{Aj}s_{Aj})$ with $\theta_{Aj}=\tau_{Aj}/(1+\tau_{Aj})$; at free trade $\theta_{Aj}=0$ and $d\theta_{Aj}=d\tau_{Aj}$, so share changes drop out and $d\log I_A=\sum_js_{Aj}\,d\tau_{Aj}$. Combining,
\begin{equation*}
\frac{dU_A}{U_A'\,I_A}=\sum_js_{Aj}\,d\tau_{Aj}-\sum_js_{Aj}(d\nu_j+d\tau_{Aj})=-m_A\sum_{j\neq A}b_{Aj}\,d\nu_j,
\end{equation*}
with $m_A=\sum_js_{Aj}$ and $b_{Aj}=s_{Aj}/m_A$. The tariff terms cancel because $\tau_{Aj}=0$ at the baseline; the cancellation is first order only. Other countries' tariffs enter $A$'s prices and income only through $d\nu$, so the formula holds for any tariff vector.
\end{proof}

\begin{proof}[Proof of \cref{prop:uniform}]
Use \cref{eq:linear} with country-specific $\eta_i$, $\rho_i$ and $\alpha_T^i$, so $s_{Aj}=\alpha_T^A(1-h_A)b_{Aj}$ and $m_A=\alpha_T^A(1-h_A)$. At $d\nu=0$, no country other than $A$ faces a price change, so $F_j=f_{Aj}(d\log s_{Aj}+d\log\tilde I_A-dt)$. In $A$, $dp_{Aj}=dt$ for every $j$, so $d\log P_{MA}=dt$, the cross-origin term vanishes, $d\log s_{Aj}=(1-\eta_A)h_A\,dt$ and $d\log\tilde I_A=m_A\,dt$. Hence
\begin{equation*}
F_j=f_{Aj}\big[(1-\eta_A)h_A+\alpha_T^A(1-h_A)-1\big]dt=-f_{Aj}\,\rho_D^A\,dt,\qquad \rho_D^A\equiv\eta_Ah_A+(1-h_A)(1-\alpha_T^A)>0,
\end{equation*}
so $F_j\leq0$, strictly if $f_{Aj}>0$. Under MML, \cref{eq:signrule} gives $d\nu_i\leq0$ for every $i$. If $A$ imports from every partner, $F<0$, and each row of the nonsingular nonnegative $(-J)^{-1}$ has a positive entry, so $d\nu_i<0$ for every $i$; if $J$ is irreducible, $(-J)^{-1}>0$ entrywise (\cref{lem:gs}), so $d\nu_i<0$ for every $i$ whenever $F\neq0$. By \cref{lem:welfare}, the welfare change $-m_A\sum_jb_{Aj}d\nu_j$ is positive, and the aggregate terms of trade improve for the same reason.
\end{proof}

\begin{proof}[Proof of \cref{prop:reversal}]
By \cref{lem:welfare}, a marginal tariff on $B$ worsens $A$'s aggregate terms of trade, and lowers its welfare to first order, if and only if $R_{A|B}>0$. It suffices to exhibit one baseline. Take the three-country model at the US-like parameters with sizes $L=(1,0.22,1)$, origin weights $\beta_{AB}=0.2$ and $\beta_{AC}=0.8$, other origin weights symmetric, and the free-trade baseline solved at $\bar\rho=3\rho_D=3.96$ and then held fixed ($b_{AB}=0.094$); this is panel (b) of \cref{fig:tot}, and panel (a) is the symmetric benchmark at the same parameters. MML holds at this baseline for $\rho\in[0.01,200]$. Since $\rho$ enters \cref{eq:linear} only through the factor $(1-\rho)$, $R_{A|B}=b'\adj(-J)F/\det(-J)$ with $\det(-J)>0$ under MML and a numerator quadratic in $\rho$. At this baseline the numerator is a positive multiple of $\rho^2-2.6695\rho-2.2568$, which has the single positive root $\rho^*=3.344$. Hence $R_{A|B}>0$ for $\rho^*<\rho\leq200$: the tariff on $B$ worsens $A$'s aggregate terms of trade and reduces its welfare to first order.
\end{proof}

\subsection{Verification}

The analytical Jacobians and tariff disturbances were checked against centered finite differences of the nonlinear model at symmetric and asymmetric baselines. Closed-form responses were also checked against direct solutions of the linear system, with MML verified at every reported threshold.
